  <script>
    let audioCtx = null;
    let oscillator = null;
    let gainNode = null;
    let isPlaying = false;
    let currentFreq = 165;
    let isSweepMode = false;
    let sweepInterval = null;
    
    // Timer variables
    let countdown = 30;
    let timerInterval = null;

    function toggleMenu() {
      document.getElementById('sideDrawer').classList.toggle('open');
      document.getElementById('overlay').classList.toggle('active');
    }

    function initAudio() {
      const AudioContext = window.AudioContext || window.webkitAudioContext;
      audioCtx = new AudioContext();

      oscillator = audioCtx.createOscillator();
      gainNode = audioCtx.createGain();

      oscillator.type = 'sawtooth';
      oscillator.frequency.setValueAtTime(currentFreq, audioCtx.currentTime);

      oscillator.connect(gainNode);
      gainNode.connect(audioCtx.destination);

      oscillator.start();
    }

    function toggleAudio() {
      if (!isPlaying) {
        startAudio();
      } else {
        stopAudio();
      }
    }

    function startAudio() {
      const container = document.getElementById('buttonContainer');
      const statusLabel = document.getElementById('statusLabel');

      if (!audioCtx) {
        initAudio();
      } else if (audioCtx.state === 'suspended') {
        audioCtx.resume();
      }

      gainNode.gain.setValueAtTime(1, audioCtx.currentTime);
      container.classList.add('playing');
      isPlaying = true;

      if (isSweepMode) {
        statusLabel.innerText = 'Sweeping frequencies active...';
        startSweep();
      } else {
        statusLabel.innerText = `Ejecting water at ${currentFreq} Hz...`;
      }

      startTimer();
    }

    function stopAudio() {
      const container = document.getElementById('buttonContainer');
      const statusLabel = document.getElementById('statusLabel');

      if (gainNode) {
        gainNode.gain.setValueAtTime(0, audioCtx.currentTime);
      }

      container.classList.remove('playing');
      statusLabel.innerText = 'Audio stopped. Tap button to restart.';
      isPlaying = false;

      stopSweep();
      resetTimer();
    }

    function updateFrequency(val) {
      currentFreq = val;
      document.getElementById('freqLabel').innerText = val + ' Hz';
      isSweepMode = false;
      clearModeButtons();

      if (oscillator && isPlaying) {
        oscillator.frequency.setValueAtTime(val, audioCtx.currentTime);
        document.getElementById('statusLabel').innerText = `Ejecting water at ${val} Hz...`;
      }
    }

    function setPreset(mode) {
      clearModeButtons();
      stopSweep();

      if (mode === 'sweep') {
        isSweepMode = true;
        document.getElementById('btn-sweep').classList.add('active');
        document.getElementById('freqLabel').innerText = 'Sweep (100–300 Hz)';
        if (isPlaying) startSweep();
      } else {
        isSweepMode = false;
        currentFreq = mode;
        document.getElementById('freqSlider').value = mode;
        document.getElementById('freqLabel').innerText = mode + ' Hz';
        document.getElementById(`btn-${mode}`).classList.add('active');
        if (oscillator && isPlaying) {
          oscillator.frequency.setValueAtTime(mode, audioCtx.currentTime);
        }
      }
    }

    function clearModeButtons() {
      document.querySelectorAll('.mode-btn').forEach(btn => btn.classList.remove('active'));
    }

    function startSweep() {
      let sweepFreq = 100;
      let direction = 1;
      
      sweepInterval = setInterval(() => {
        if (!isPlaying) return;
        sweepFreq += direction * 5;
        if (sweepFreq >= 300) direction = -1;
        if (sweepFreq <= 100) direction = 1;
        
        if (oscillator) {
          oscillator.frequency.setValueAtTime(sweepFreq, audioCtx.currentTime);
        }
      }, 50);
    }

    function stopSweep() {
      if (sweepInterval) clearInterval(sweepInterval);
    }

    function startTimer() {
      countdown = 30;
      updateTimerDisplay();
      
      timerInterval = setInterval(() => {
        countdown--;
        updateTimerDisplay();

        if (countdown <= 0) {
          stopAudio();
          document.getElementById('statusLabel').innerText = 'Ejection cycle complete!';
        }
      }, 1000);
    }

    function resetTimer() {
      if (timerInterval) clearInterval(timerInterval);
      countdown = 30;
      updateTimerDisplay();
    }

    function updateTimerDisplay() {
      const secs = countdown < 10 ? `0${countdown}` : countdown;
      document.getElementById('timerDisplay').innerText = `00:${secs}`;
    }
  </script>
